    \begin{equation}
        \Delta L = \frac{L_0}{\gamma}=L_0 \sqrt{1 - \beta^2} \tag{length contraction}
    \end{equation}
    \begin{equation}
        u = \frac{u' + v}{1 + u'(\frac{v}{c^2})} \notag
    \end{equation}
Where $u'$ is the velocity of an object from the moving frame.
    $v$ is the velocity of the stationary frame.
    $u$ is the velocity of the object in respect to the stationary frame.
